## 3. 主组件骨架

```vue
<template>
  <div class="space-browser">
    <header class="space-browser__topbar">
      <h1>书的世界</h1>
      <input v-model="query" placeholder="搜索书 / 主题 / 地点" />
    </header>

    <main class="space-browser__stage">
      <Canvas3DScene
        :items="filteredItems"
        @select="handleSelect"
      />

      <SpaceBrowserInfoCard
        v-if="selectedItem"
        :item="selectedItem"
        @open="openDetail"
      />
    </main>
  </div>
</template>

<script setup>
import { computed, ref } from 'vue'
import { spaceItems } from './spaceBrowserData'
import SpaceBrowserInfoCard from './SpaceBrowserInfoCard.vue'

const query = ref('')
const selectedId = ref(null)

const filteredItems = computed(() => {
  const q = query.value.trim().toLowerCase()
  if (!q) return spaceItems
  return spaceItems.filter(item =>
    [item.title, item.subtitle, ...(item.tags || [])].some(v =>
      String(v).toLowerCase().includes(q)
    )
  )
})

const selectedItem = computed(() =>
  spaceItems.find(item => item.id === selectedId.value) || null
)

function handleSelect(item) {
  selectedId.value = item.id
}

function openDetail() {
  if (selectedItem.value?.targetUrl) {
    window.location.href = selectedItem.value.targetUrl
  }
}
</script>
```
